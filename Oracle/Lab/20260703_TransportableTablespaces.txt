ADMIN USER pdbadmin IDENTIFIED BY "0racl3##" ROLES=(CONNECT)
FILE_NAME_CONVERT=('/u01/oradata/ORCLDB/pdbseed/','/u01/oradata/ORCLDB/pdb2/');
ALTER PLUGGABLE DATABASE PDB2 open;
ALTER PLUGGABLE DATABASE PDB2 SAVE STATE;
