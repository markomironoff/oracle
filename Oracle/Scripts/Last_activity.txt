SELECT *
FROM (
  SELECT sql_id,
         TO_CHAR(last_active_time,'YYYY-MM-DD HH24:MI:SS') AS last_active_time,
         parsing_schema_name,
         module,
         action,
         command_type,
         sql_text
  FROM   v$sql
  WHERE  parsing_schema_name IS NOT NULL
    AND  UPPER(parsing_schema_name) NOT IN ('SYS','TS_MONITOR','PERFSTAT','EXFSYS','ORACLE_OCM')
  ORDER  BY last_active_time DESC NULLS LAST
)
WHERE ROWNUM <= 20;
