set lines 220 pages 100
col name format a15
col state format a10
col type format a8

select
    group_number,
    name,
    state,
    type,
    round(total_mb/1024,2) as total_gb,
    round(free_mb/1024,2) as free_gb,
    round(usable_file_mb/1024,2) as usable_free_gb,
    round((total_mb - free_mb)/1024,2) as used_gb,
    round(free_mb * 100 / nullif(total_mb,0), 2) as pct_free,
    round((total_mb - free_mb) * 100 / nullif(total_mb,0), 2) as pct_used
from v$asm_diskgroup
order by name;
